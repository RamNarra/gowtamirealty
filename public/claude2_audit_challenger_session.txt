def main():
    session_logs = glob.glob("/home/p4cketsn1ff3r/.claude/projects/-home-p4cketsn1ff3r-Projects-Active-KAVACH-AI/*.jsonl")
    if not session_logs:
        print("No session files found.")
        return

    session_logs.sort(key=lambda p: os.path.getmtime(p), reverse=True)
    latest_file = session_logs[0]
    filename = os.path.basename(latest_file)
    session_uuid = filename.replace(".jsonl", "")

    mtime = datetime.fromtimestamp(os.path.getmtime(latest_file))
    print(f"Latest session: {session_uuid} ({mtime.strftime('%Y-%m-%d %H:%M:%S')})")

    lines = []
    with open(latest_file, "r") as f:
        for line in f:
            try:
                lines.append(json.loads(line))
            except Exception:
                pass

    if not lines:
        print("Empty session file.")
        return

    user_prompt = ""
    prompt_idx = 0
    for i, d in enumerate(lines):
        if d.get("type") == "user" and not d.get("isMeta"):
            c = d.get("message", {}).get("content")
            if isinstance(c, str):
                user_prompt = c
                prompt_idx = i
                break
            elif isinstance(c, list) and not any(x.get("type") == "tool_result" for x in c):
                user_prompt = str(c)
                prompt_idx = i
                break

    out = []
    out.append("=" * 80)
    out.append(f"CLAUDE 2 SESSION TRACE - {session_uuid}")
    out.append("=" * 80)
    out.append(f"Timestamp: {mtime.isoformat()}")
    out.append(f"Source: {latest_file}\n")
    out.append("=" * 80)
    out.append("USER PROMPT")
    out.append("=" * 80)
    out.append(user_prompt)
    out.append("\n" + "=" * 80)
    out.append("FULL VERBOSE EXECUTION TRACE")
    out.append("=" * 80)

    step_num = 1
    final_output = ""
    thinking_count = 0
    tool_count = 0

    for i in range(prompt_idx + 1, len(lines)):
        d = lines[i]
        t = d.get("type")
        if t == "assistant":
            for c in d.get("message", {}).get("content", []):
                ct = c.get("type")
                if ct == "thinking":
                    t_str = c.get("thinking", "")
                    if t_str:
                        out.append(f"\n--- [Step {step_num} | THINKING] ---\n{t_str}")
                        step_num += 1
                        thinking_count += 1
                elif ct == "tool_use":
                    tool_count += 1
                    inp = json.dumps(c.get("input", {}), indent=2)
                    out.append(f"\n--- [Step {step_num} | TOOL: {c.get('name')}] ---\n{inp}")
                    step_num += 1
                elif ct == "text":
                    txt = c.get("text", "")
                    if txt:
                        out.append(f"\n--- [Step {step_num} | ASSISTANT] ---\n{txt}")
                        step_num += 1
                        final_output = txt
        elif t == "user":
            for c in d.get("message", {}).get("content", []):
                if c.get("type") == "tool_result":
                    res = c.get("content", "")
                    if isinstance(res, list):
                        res = "\n".join(str(r) for r in res)
                    out.append(f"\n[TOOL RESULT | {'ERROR' if c.get('is_error') else 'SUCCESS'}]\n{res}")

    full_text = "\n".join(out)
    out_filename = f"claude2_session_{session_uuid[:8]}.txt"

    for proj in PROJECTS:
        pub_path = os.path.join(proj, "public", out_filename)
        with open(pub_path, "w") as f:
            f.write(full_text)
        print(f"Saved {len(full_text)} bytes to {pub_path}")

    print(f"Sync complete: {step_num} steps, {thinking_count} thinking blocks, {tool_count} tool calls.")
